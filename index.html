// Estado de la Aplicación (Extendido con datos de brazada) let estado = { activo: false, tiempoInicio: 0, intervaloGeneral: null, nadadores: [] }; let chartInstancia = null; // Inicialización document.getElementById('fechaTorneo').valueAsDate = new Date(); generarCarriles(); inicializarGrafica(); function generarCarriles() { const cantidad = parseInt(document.getElementById('numNadadores').value) || 1; const container = document.getElementById('gridCarriles'); container.innerHTML = ''; estado.nadadores = []; for (let i = 1; i <= cantidad; i++) { estado.nadadores.push({ id: i, nombre: `Nadador ${i}`, tiemposParciales: [], // ms brazadasParciales: [], // Conteo de ciclos/brazadas por parcial de 50m finalizado: false }); container.innerHTML += `
CARRIL ${i} Listo
00:00.00
Sin parciales
`; } actualizarReporte(); } function actualizarNombre(id, nuevoNombre) { const nadador = estado.nadadores.find(n => n.id === id); if (nadador) nadador.nombre = nuevoNombre; actualizarReporte(); } function iniciarTodos() { if (estado.activo) return; estado.activo = true; estado.tiempoInicio = Date.now(); estado.intervaloGeneral = setInterval(() => { const transcurrido = Date.now() - estado.tiempoInicio; document.getElementById('relojGeneral').innerText = formatearTiempo(transcurrido); estado.nadadores.forEach(n => { if (!n.finalizado) { document.getElementById(`display-${n.id}`).innerText = formatearTiempo(transcurrido); } }); }, 10); } function marcarSplit(id) { if (!estado.activo) return; const nadador = estado.nadadores.find(n => n.id === id); if (nadador && !nadador.finalizado) { const tiempoActual = Date.now() - estado.tiempoInicio; const numBrazadas = parseInt(document.getElementById(`inputBrazada-${id}`).value) || 0; nadador.tiemposParciales.push(tiempoActual); nadador.brazadasParciales.push(numBrazadas); renderizarSplitsCarril(nadador); actualizarReporte(); } } function detenerNadador(id) { if (!estado.activo) return; const nadador = estado.nadadores.find(n => n.id === id); if (nadador && !nadador.finalizado) { const tiempoFinal = Date.now() - estado.tiempoInicio; const numBrazadas = parseInt(document.getElementById(`inputBrazada-${id}`).value) || 0; nadador.tiemposParciales.push(tiempoFinal); nadador.brazadasParciales.push(numBrazadas); nadador.finalizado = true; document.getElementById(`status-${id}`).innerText = "Finalizado"; document.getElementById(`status-${id}`).className = "text-xs px-2 py-0.5 rounded bg-emerald-900 text-emerald-300"; renderizarSplitsCarril(nadador); actualizarReporte(); } } function renderizarSplitsCarril(nadador) { const container = document.getElementById(`splits-summary-${nadador.id}`); let html = ''; let ultimoTiempo = 0; nadador.tiemposParciales.forEach((t, index) => { const lap = t - ultimoTiempo; const brazadas = nadador.brazadasParciales[index] || 0; html += `
#${index + 1} (${(index + 1) * 50}m): ${formatearTiempo(t)} [Brz: ${brazadas}]
`; ultimoTiempo = t; }); container.innerHTML = html; } function reiniciarTodo() { clearInterval(estado.intervaloGeneral); estado.activo = false; estado.tiempoInicio = 0; document.getElementById('relojGeneral').innerText = "00:00.00"; generarCarriles(); } function formatearTiempo(ms) { const min = Math.floor(ms / 60000); const seg = Math.floor((ms % 60000) / 1000); const cent = Math.floor((ms % 1000) / 10); return `${String(min).padStart(2, '0')}:${String(seg).padStart(2, '0')}.${String(cent).padStart(2, '0')}`; } // Generación de Reportes con Métricas de Eficiencia Táctica function actualizarReporte() { const fecha = document.getElementById('fechaTorneo').value; const piscina = document.getElementById('longitudPiscina').value; const prueba = document.getElementById('distanciaPrueba').value; document.getElementById('repFecha').innerText = fecha; document.getElementById('repPiscina').innerText = `${piscina} metros`; document.getElementById('repPrueba').innerText = `${prueba} metros`; document.getElementById('repHora').innerText = new Date().toLocaleTimeString(); const contenedorTablas = document.getElementById('tablasResultados'); contenedorTablas.innerHTML = ''; const datasetsGrafica = []; estado.nadadores.forEach(nadador => { if (nadador.tiemposParciales.length === 0) return; let tablaHTML = `

${nadador.nombre} (Carril ${nadador.id})

`; let tiempoAnterior = 0; const indicesSEI = []; nadador.tiemposParciales.forEach((tAcumulado, idx) => { const tiempoLapSeg = (tAcumulado - tiempoAnterior) / 1000; const distanciaLap = 50; const brazadas = nadador.brazadasParciales[idx] || 1; // Evita división por cero // Cálculos Cinematices const msVelocidad = tiempoLapSeg > 0 ? (distanciaLap / tiempoLapSeg) : 0; const strokeLength = distanciaLap / brazadas; // SL (m/ciclo) const strokeRate = (brazadas / tiempoLapSeg) * 60; // SR (Ciclos/min) const sei = msVelocidad * strokeLength; // SEI = v * SL indicesSEI.push(sei.toFixed(2)); tablaHTML += ` `; tiempoAnterior = tAcumulado; }); tablaHTML += `
Tramo Acumulado Lap (s) Brazadas Velocidad (m/s) SL (m/ciclo) SR (SPM) SEI (Eficiencia)
${(idx + 1) * 50}m ${formatearTiempo(tAcumulado)} ${tiempoLapSeg.toFixed(2)}s ${brazadas} ${msVelocidad.toFixed(2)} m/s ${strokeLength.toFixed(2)} m ${strokeRate.toFixed(1)} SPM ${sei.toFixed(2)}
`; contenedorTablas.innerHTML += tablaHTML; const colores = ['#0284c7', '#059669', '#d97706', '#dc2626', '#7c3aed', '#db2777', '#4b5563', '#0891b2']; datasetsGrafica.push({ label: `${nadador.nombre} (SEI)`, data: indicesSEI, borderColor: colores[(nadador.id - 1) % colores.length], backgroundColor: colores[(nadador.id - 1) % colores.length], tension: 0.2, fill: false }); }); actualizarGrafica(datasetsGrafica); } function inicializarGrafica() { const ctx = document.getElementById('chartVelocidad').getContext('2d'); chartInstancia = new Chart(ctx, { type: 'line', data: { labels: ['50m', '100m', '150m', '200m'], datasets: [] }, options: { responsive: true, maintainAspectRatio: false, scales: { y: { title: { display: true, text: 'Índice SEI (v × SL)' }, suggestedMin: 1.0, suggestedMax: 4.0 }, x: { title: { display: true, text: 'Parciales de Carrera' } } } } }); } function actualizarGrafica(datasets) { if (!chartInstancia) return; let maxSplits = 0; datasets.forEach(d => { if (d.data.length > maxSplits) maxSplits = d.data.length; }); const labels = []; for (let i = 1; i <= maxSplits; i++) { labels.push(`${i * 50}m`); } chartInstancia.data.labels = labels; chartInstancia.data.datasets = datasets; chartInstancia.update(); } function guardarEvento() { const fecha = document.getElementById('fechaTorneo').value; const prueba = document.getElementById('distanciaPrueba').value; const idGuardado = `SwimData_${fecha}_${prueba}m_${Date.now()}`; const datosAguardar = { fecha: fecha, piscina: document.getElementById('longitudPiscina').value, prueba: prueba, nadadores: estado.nadadores }; localStorage.setItem(idGuardado, JSON.stringify(datosAguardar)); alert(`Control técnico guardado exitosamente con ID: ${idGuardado}`); }
